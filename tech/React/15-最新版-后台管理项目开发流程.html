  },

    // payload = { title: '', path: ''}

      if (payload.path === "/dashboard") return;

      let curIndex = state.tabList.findIndex((item) => item.path === payload.path);
      console.log(curIndex);
      if (curIndex !== -1) {
        //如果重复
        state.activeIndex = curIndex
      } else {
        //不重复
        state.tabList = [...state.tabList, payload];
        state.activeIndex = state.tabList.length - 1;
      }
    },
    delTab: (state,{payload})=>{
      let maxIndex = state.tabList.length - 1
      let curIndex = state.tabList.findIndex((item) => item.path === payload.path);
      state.tabList.splice(curIndex,1)
      //删除项索引小于当前项
      if (curIndex < state.activeIndex){
        state.activeIndex--
        return;
      }
      //删除项就是当前项，且不是最后一项
      if (curIndex === state.activeIndex && curIndex !== maxIndex){
        return;
      }
      //删除最后一项
      if (curIndex === maxIndex){
          state.activeIndex = maxIndex - 1;
      }
    }
  },
});
